157 void animate(double delta) {
158 this->frame_timer += delta;
159
160 if (this->frame_timer >= this->frame_duration) {
161 this->frame_timer = 0;
162 this->frame_rect.left += this->frame_rect.width;
163 if (this->frame_rect.left >=
164 this->sprite.getTexture()->getSize().x - 5) {
165 this->frame_rect.left = 0;
166 done_once = true;

168 this->sprite.setTextureRect(this->frame_rect);
169 }
