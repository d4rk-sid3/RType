145 void setFrameRect(int width, int height) {
146 this->frame_rect = sf::IntRect(0, 0, width, height);
147 this->sprite.setTextureRect(this->frame_rect);
148 }
149
155 void animate(double delta) {
156 this->frame_timer += delta;
157
158 if (this->frame_timer >= this->frame_duration) {
159 this->frame_timer = 0;
160 this->frame_rect.left += this->frame_rect.width;
161 if (this->frame_rect.left >=
162 this->sprite.getTexture()->getSize().x - 5) {
163 this->frame_rect.left = 0;
164 done_once = true;
165 }
166 this->sprite.setTextureRect(this->frame_rect);
167 }
168 }
