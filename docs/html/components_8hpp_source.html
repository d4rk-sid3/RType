114 void animate(double delta) {
115 this->frame_timer += delta;
116
117 if (this->frame_timer >= this->frame_duration) {
118 this->frame_timer = 0;
119 this->frame_rect.left += this->frame_rect.width;
120 if (this->frame_rect.left >= this->sprite.getTexture()->getSize().x - 5) {
121 this->frame_rect.left = 0;
122 done_once = true;

124 }
125 this->sprite.setTextureRect(this->frame_rect);
126 }
